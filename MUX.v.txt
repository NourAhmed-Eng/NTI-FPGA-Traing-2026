module MUX #( parameter WIDTH = 5 )

(
    //INPUTS
    input  wire [WIDTH -1:0] in0,
    input  wire [WIDTH -1:0] in1,
    input  wire  sel,


    //OUTPUTS
    output reg [WIDTH -1:0] mux_out

);

    always @(*) begin

    if (! sel )
    mux_out = in0;

    else
    mux_out = in1;

    end


endmodule


module MUX_tb;

    parameter WIDTH = 5;

    reg  [WIDTH-1:0] in0;
    reg  [WIDTH-1:0] in1;
    reg sel;


    wire [WIDTH-1:0] mux_out;

    MUX #(.WIDTH(WIDTH)) dut (
        .in0(in0),
        .in1(in1),
        .sel(sel),
        .mux_out(mux_out)
    );

    initial begin
    
        $monitor("At time %0t sel=%b in0=%b in1=%b, mux_out=%b",
         $time, sel, in0, in1, mux_out);

        sel = 1'b0; in0 = 5'b10101; in1 = 5'b00000;
        #5;

        sel = 1'b0; in0 = 5'b01010; in1 = 5'b00000;
        #5;

        sel = 1'b1; in0 = 5'b00000; in1 = 5'b10101;
        #5;

        sel = 1'b1; in0 = 5'b00000; in1 = 5'b01010;
        #5;

        $finish;
    end

endmodule
